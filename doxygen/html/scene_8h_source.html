422 void rope(point *circ, UINT nspring, point *p0, point *p1, int neg = 1)
423 {
424 if (circ->radius == 0){
425 return;
426 }
427 float rght = circ->x + circ->radius;
428 float lft = circ->x - circ->radius;
429 float top = circ->y - circ->radius;
430 float bttm = circ->y + circ->radius;
431 if (((p0->x <= lft && p1->x >= lft) ||
432 (p1->x <= lft && p0->x >= lft) ||
433 (p0->x <= rght && p1->x >= rght) ||
434 (p1->x <= rght && p0->x >= rght)) ||
435 ((p0->y <= top && p1->y >= top) ||
436 (p1->y <= top && p0->y >= top) ||
437 (p0->y <= bttm && p1->y >= bttm) ||
438 (p1->y <= bttm && p0->y >= bttm))){
439 point virt;
440 if (p1->x == p0->x){
441 virt.x = p0->x;
442 virt.y = circ->y;
443 }
444 else{
445 float a = (p1->y - p0->y)/(p1->x - p0->x);
446 float b = p0->y - a * p0->x;
447 virt.x = ((circ->x + a*circ->y - a * b)/(a*a + 1));
448 virt.y = (a * circ->x + a*a*circ->y + b)/(a*a + 1);
449 }
450 if (isInside(p0, p1, &virt)){
451 float d0 = distance(p0->x, p0->y, virt.x, virt.y) /
452 distance(p0->x, p0->y, p1->x, p1->y);
453 float d1 = distance(p1->x, p1->y, virt.x, virt.y) /
454 distance(p0->x, p0->y, p1->x, p1->y);
455 float td = d0 + d1;

457 virt.vx = p0->vx + (p1->vx-p0->vx) * rd;
458 virt.fx = p0->fx + (p1->fx-p0->fx) * rd;
459 virt.vy = p0->vy + (p1->vy-p0->vy) * rd;
460 virt.fy = p0->fy + (p1->fy-p0->fy) * rd;
461 virt.mass = (p0->mass + p1->mass) / 2;
462 virt.n = nspring;
463 float dx = virt.x - circ->x;
464 float dy = virt.y - circ->y;
465 float dsq = dx * dx + dy * dy;
466 float rsq = circ->radius * circ->radius;
467 if (dsq <= rsq){
468 //dumpthis();
